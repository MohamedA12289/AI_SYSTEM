```python
import time

def show_instructions():
    print("You are standing at a crossroads.")
    print("You can go North, South, East or West.")
    print("Type 'go [direction]' to move in that direction.")
    print("Type 'quit' to exit the game.")

def play_game():
    show_instructions()
    
    current_room = "start"
    while True:
        command = input("> ").lower().strip()
        
        if command == "quit":
            print("Thanks for playing!")
            break
        
        if command.startswith("go"):
            direction = command.split()[1]
            if direction == "north" and current_room == "start":
                current_room = "north_room"
                print("You are now in the North Room.")
            elif direction == "south" and current_room == "start":
                current_room = "south_room"
                print("You are now in the South Room.")
            elif direction == "east" and current_room == "start":
                current_room = "east_room"
                print("You are now in the East Room.")
            elif direction == "west" and current_room == "start":
                current_room = "west_room"
                print("You are now in the West Room.")
            else:
                print("You can't go that way from here.")
        else:
            print("I don't understand that command.")

if __name__ == "__main__":
    play_game()
```
